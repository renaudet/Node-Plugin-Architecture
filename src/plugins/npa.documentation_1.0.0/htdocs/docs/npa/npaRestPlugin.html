npa.rest — REST Client Plugin

Overview

The npa.rest plugin provides a unified HTTP/HTTPS client for making outbound REST API calls from server-side plugin code. It wraps the axios library, handles self-signed certificates, optional Basic-Auth credentials, and custom headers, and exposes a single entry point:
let rest = plugin.getService('rest');
REST call counts are pushed to the rest.calls telemetry dimension every 30 seconds.

Service Name

'rest'

The restContext Object

All API methods accept a single restContext descriptor:
 1. {
 2.   "host": "api.example.com",        // required
 3.   "port": 443,                    // optional, omit for default 80/443
 4.   "uri": "/api/v1/resource",        // required
 5.   "secured": true,                // use HTTPS when true
 6.   "acceptCertificate": true,     // accept self-signed certs
 7.   "method": "GET",                // GET | POST | PUT | DELETE
 8.   "username": "user",             // optional Basic-Auth
 9.   "password": "secret",            // optional Basic-Auth
10.   "payload": { ... },            // optional body for POST/PUT
11.   "options": {                   // optional axios options
12.     "headers": { "Content-Type": "application/json" }
13.   }
14. }

Method

performRestApiCall(restContext, callback)

Dispatches the call to GET, POST, PUT or DELETE based on restContext.method. The callback follows the Node.js error-first convention:
function(err, response) where response is the raw axios response object (access the body via response.data).

Usage Examples

GET request

 1. let rest = plugin.getService('rest');
 2. let ctx = {
 3.   "host": "api.example.com",
 4.   "uri": "/data/items",
 5.   "secured": false,
 6.   "method": "GET"
 7. };
 8. 
 9. rest.performRestApiCall(ctx, function(err, response) {
10.   if(err) {
11.     plugin.error('Call failed: ' + err);
12.   } else {
13.     let items = response.data;
14.     plugin.info('Received ' + items.length + ' items');
15.   }
16. });

POST request with JSON payload

 1. let rest = plugin.getService('rest');
 2. let ctx = {
 3.   "host": "api.example.com",
 4.   "uri": "/data/items",
 5.   "secured": true,
 6.   "acceptCertificate": true,
 7.   "method": "POST",
 8.   "payload": { "name": "newItem", "value": 42 }
 9. };
10. 
11. rest.performRestApiCall(ctx, function(err, response) {
12.   if(err) {
13.     plugin.error('POST failed: ' + err);
14.   } else {
15.     plugin.info('Created: ' + JSON.stringify(response.data));
16.   }
17. });

HTTPS with Basic Authentication

 1. let rest = plugin.getService('rest');
 2. let ctx = {
 3.   "host": "secure.example.com",
 4.   "port": 8443,
 5.   "uri": "/protected/resource",
 6.   "secured": true,
 7.   "acceptCertificate": true,
 8.   "method": "GET",
 9.   "username": "admin",
10.   "password": "s3cr3t"
11. };
12. 
13. rest.performRestApiCall(ctx, function(err, response) {
14.   if(!err) plugin.info(JSON.stringify(response.data));
15. });

Notes

  • The response body is always at response.data (axios convention).
  • When acceptCertificate: true, the HTTPS agent is configured with rejectUnauthorized: false — use only with trusted internal endpoints.
  • REST call counts are reported to the rest.calls telemetry dimension every 30 seconds.
  • Custom headers (e.g. Authorization: Bearer ...) can be passed via options.headers.
